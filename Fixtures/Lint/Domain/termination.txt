// Linted as a file of the Domain folder: the termination rule reaches it as it reaches the root of
// the library. A doc comment is not a call, so the line before assertValid gives no diagnostic.
func terminate() {
    fatalError("x") // expect: no_process_termination
    preconditionFailure("x") // expect: no_process_termination
    assertionFailure("x") // expect: no_process_termination
    precondition(false) // expect: no_process_termination
    assert(false) // expect: no_process_termination
    dispatchPrecondition(condition: .onQueue(.main)) // expect: no_process_termination
    abort() // expect: no_process_termination
    exit(1) // expect: no_process_termination
    // fatalError("x")
    let quoted = "fatalError(\"x\")"
    let multiline = """
    exit(1)
    """
    assertValid()
}

/// exit(1)
func assertValid() {}

func terminateByName() {
    Swift.fatalError("x") // expect: no_process_termination
    Darwin.exit(0) // expect: no_process_termination
    fatalError ("x") // expect: no_process_termination
    if true { exit(1) } else { abort() } // expect: no_process_termination no_process_termination
    app.exit(1) // expect: no_process_termination
    _exit(1)
    raise(9)
    exitCode(1)
    myexit(1)
    exit_now(1)
    Assert(true)
}
