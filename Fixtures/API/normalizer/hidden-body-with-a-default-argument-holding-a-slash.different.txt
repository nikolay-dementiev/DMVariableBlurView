# expect: different
# A hidden declaration is dropped by its indentation together with its body, whatever the
# body holds. The public member after it stays at the indentation of its type, so
# its removal is an API change. Its default argument holds a bare regex literal with a brace.
--- A ---
public struct First {
  package enum Hidden {
    package static func matches(_ pattern: Regex<Substring> = /\{/) -> Swift.Bool
  }
  public func removedLater()
  public func keep()
}
