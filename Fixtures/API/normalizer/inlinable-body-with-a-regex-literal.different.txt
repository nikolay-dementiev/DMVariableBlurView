# A hidden declaration is dropped by its indentation together with its body, whatever the
# body holds. The public member after it stays at the indentation of its type, so
# its removal is an API change. Its body holds an extended regex literal with a brace.

  @inlinable internal func helper() -> Swift.Bool {
    return "x".contains(#/\{/#)
  }
  public func visible()
}
